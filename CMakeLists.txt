cmake_minimum_required(VERSION 3.16)
project(ComputerVisionLab2 VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# OpenCV 5 renamed features2d and split calib3d into several modules. Discover
# the installed version first, then link only the modules this project uses.
find_package(OpenCV REQUIRED)
if(OpenCV_VERSION VERSION_LESS 4.4)
    message(FATAL_ERROR "OpenCV 4.4 or newer is required for cv::SIFT.")
endif()
set(LAB2_OPENCV_MODULES opencv_core opencv_imgproc opencv_video opencv_videoio opencv_imgcodecs opencv_highgui)
if(OpenCV_VERSION VERSION_GREATER_EQUAL 5.0)
    list(APPEND LAB2_OPENCV_MODULES opencv_features opencv_geometry)
else()
    list(APPEND LAB2_OPENCV_MODULES opencv_features2d opencv_calib3d)
endif()
foreach(module IN LISTS LAB2_OPENCV_MODULES)
    if(NOT TARGET ${module})
        message(FATAL_ERROR "This OpenCV installation does not provide required module ${module}.")
    endif()
endforeach()
add_library(planar_tracker src/planar_tracker.cpp)
target_include_directories(planar_tracker PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(planar_tracker PUBLIC ${LAB2_OPENCV_MODULES})
target_compile_options(planar_tracker PRIVATE -Wall -Wextra -Wpedantic)

add_executable(lab2_tracker src/main.cpp)
target_link_libraries(lab2_tracker PRIVATE planar_tracker)
target_compile_options(lab2_tracker PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_tracker tests/test_tracker.cpp)
    target_link_libraries(test_tracker PRIVATE planar_tracker)
    add_test(NAME tracker_geometry_and_recovery COMMAND test_tracker)
endif()
